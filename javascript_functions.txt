Arrow functions = arrow functions provide shorter way to write functions
Supported for single-line expressions
() => { ... }

    // statement
}

  console.log("hoo");
};
demo();

const calculateTotal = (price,quantity) =>
  {
    return price * quantity;
    
  };

callbacks functions - A callback function is a function that is passed as an argument to another function and is executed (called) inside that function.

function welcome(callback) {
    console.log("Welcome");
    callback();
}

function message() {
    console.log("Good Morning");
}

welcome(message);
